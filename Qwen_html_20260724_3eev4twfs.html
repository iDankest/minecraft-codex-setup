retry.ts· typescript
export async function retry<T>(
  fn: () => Promise<T>,
  tries = 5
): Promise<T> {
  for (let i = 0; i < tries; i++) {
    try { return await fn(); }
    catch (e) {
      if (i === tries - 1) throw e;
      const wait = 2 ** i * 200
        + Math.random() * 100;
      await new Promise(r =>
        setTimeout(r, wait));
    }
  }
}
